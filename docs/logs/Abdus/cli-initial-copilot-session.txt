````
Selection atomically:

2. Revalidates the open request, pending proposal, date, and category.
3. Generates one engagement ID.
4. Holds service payment plus levy without taking available balance below -180.
5. Marks the chosen proposal `SELECTED`.
6. Closes the request.
7. Rejects competing pending proposals.
8. Creates one `AGREED` engagement.

```java
Result<EngagementView> confirmCompletion(
    EngagementId engagementId,
    MemberId actorId
);

Only the requester and helper may confirm on or after the engagement date. The first distinct confirmation records no transaction. ### Cancellation

```java
Result<EngagementView> cancelEngagement(
    EngagementId engagementId,
    MemberId actorId
);
```

Either participant may cancel only while `AGREED` and only when `today < engagementDate`. The immediately preceding calendar date is late. Success records the actor, date, and late flag, releases the hold, and writes no transaction.
````

```
package ie.ucd.meitheal.cli;

import java.util.ArrayList;
import java.util.List;

public final class CommandParser {
    public ParsedCommand parse(String line) {
        if (line == null) {
            throw new IllegalArgumentException("Command line must not be null.");
        }

        List<String> tokens = new ArrayList<>();
        StringBuilder token = new StringBuilder();
        boolean inQuotes = false;
        boolean tokenStarted = false;

        for (int index = 0; index < line.length(); index++) {
            char current = line.charAt(index);
            if (inQuotes) {
                if (current == '"') {
                    inQuotes = false;
                } else if (current == '\\') {
                    if (index + 1 >= line.length()) {
                        throw syntaxError("dangling escape in quoted argument");
                    }
                    char escaped = line.charAt(++index);
                    if (escaped != '"' && escaped != '\\') {
                        throw syntaxError("invalid escape '\\" + escaped + "' in quoted argument");
                    }
                    token.append(escaped);
                } else {
                    token.append(current);
                }
                tokenStarted = true;
            } else if (Character.isWhitespace(current)) {
                if (tokenStarted) {
                    tokens.add(token.toString());
                    token.setLength(0);
                    tokenStarted = false;
                }
            } else if (current == '"') {
                inQuotes = true;
                tokenStarted = true;
            } else {
                token.append(current);
                tokenStarted = true;
            }
        }

        if (inQuotes) {
            throw syntaxError("unmatched double quote");
        }
        if (tokenStarted) {
            tokens.add(token.toString());
        }
        if (tokens.isEmpty()) {
            throw syntaxError("missing command name");
        }

        return new ParsedCommand(tokens.get(0), tokens.subList(1, tokens.size()));
    }

    private IllegalArgumentException syntaxError(String detail) {
        return new IllegalArgumentException("Invalid command syntax: " + detail + ".");
    }
}
```

```
package ie.ucd.meitheal.cli;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.Reader;
import java.io.Writer;
import java.util.Objects;

public final class MeithealCli {
    private final BufferedReader input;
    private final Writer output;
    private final CommandParser parser;

    public MeithealCli(Reader input, Writer output) {
        this.input = new BufferedReader(Objects.requireNonNull(input, "input"));
        this.output = Objects.requireNonNull(output, "output");
        this.parser = new CommandParser();
    }

    public void run() throws IOException {
        String line;
        while ((line = input.readLine()) != null) {
            if (line.isBlank()) {
                continue;
            }

            try {
                ParsedCommand command = parser.parse(line);
                if ("exit".equals(command.name())) {
                    if (command.arguments().isEmpty()) {
                        return;
                    } else {
                        writeArgumentCountError(command);
                    }
                } else if ("help".equals(command.name())) {
                    if (command.arguments().isEmpty()) {
                        writeHelp();
                    } else {
                        writeArgumentCountError(command);
                    }
                } else {
                    writeError("Unknown command '" + command.name()
                            + "'. Type 'help' for available commands.");
                }
            } catch (IllegalArgumentException exception) {
                writeError(exception.getMessage());
            }
        }
    }

    private void writeArgumentCountError(ParsedCommand command) throws IOException {
        writeError("Command '" + command.name() + "' expects 0 arguments but got "
                + command.arguments().size() + ".");
    }

    private void writeHelp() throws IOException {
        output.write("Supported commands:\n");
        output.write("  help - Show available commands.\n");
        output.write("  exit - Exit the CLI.\n");
        output.flush();
    }

    private void writeError(String message) throws IOException {
        output.write("ERROR " + message + "\n");
        output.flush();
    }
}
```

```
package ie.ucd.meitheal.cli;

import static org.junit.jupiter.api.Assertions.assertEquals;

import java.io.StringReader;
import java.io.StringWriter;
import org.junit.jupiter.api.Test;

class MeithealCliTest {
    @Test
    void scriptedErrorsAndBlankLinesContinueToValidCommands() throws Exception {
        StringWriter output = new StringWriter();

        new MeithealCli(new StringReader(
                "\nunknown\nhelp extra\nhelp \"unfinished\nhelp\n"), output).run();

        assertEquals("""
                ERROR Unknown command 'unknown'. Type 'help' for available commands.
                ERROR Command 'help' expects 0 arguments but got 1.
                ERROR Invalid command syntax: unmatched double quote.
                Supported commands:
                  help - Show available commands.
                  exit - Exit the CLI.
                """, output.toString());
    }

    @Test
    void wrongExitArgumentCountAllowsTheNextCommand() throws Exception {
        StringWriter output = new StringWriter();

        new MeithealCli(new StringReader("exit now\nhelp\nexit\n"), output).run();

        assertEquals("""
                ERROR Command 'exit' expects 0 arguments but got 1.
                Supported commands:
                  help - Show available commands.
                  exit - Exit the CLI.
                """, output.toString());
    }

    @Test
    void exitStopsProcessingSubsequentInput() throws Exception {
        StringWriter output = new StringWriter();

        new MeithealCli(new StringReader("help\nexit\nhelp\nunknown\n"), output).run();

        assertEquals("""
                Supported commands:
                  help - Show available commands.
                  exit - Exit the CLI.
                """, output.toString());
    }

    @Test
    void eofTerminatesWithoutOutput() throws Exception {
        StringWriter output = new StringWriter();

        new MeithealCli(new StringReader(""), output).run();

        assertEquals("", output.toString());
    }
}
```
